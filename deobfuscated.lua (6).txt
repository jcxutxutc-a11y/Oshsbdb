local function genPath(p, angle, steps, stepLen, jitter, depth, out, scale)

	for i = 1, steps do

		p = p + Vector2.new(math.cos(angle), math.sin(angle)) * stepLen * (0.7 + math.random() * 0.6)
		pts[#pts + 1] = p
		if depth > 0 and i > 2 and math.random() < 0.22 then
			local side = math.random() < 0.5 and -1 or 1
			genPath(p, angle + side * (0.5 + math.random() * 0.6), math.max(3, steps - i - 2), stepLen * 0.8, jitter * 1.2, depth - 1, out, scale * 0.55)
		end
	end
	out[#out + 1] = {pts = pts, scale = scale}
end

	end
	lightClouds(false)

	tw(SkyGlow, big and 1.4 or 0.5, {BackgroundTransparency = 1})
	task.wait(0.6)
	BoltLayer:ClearAllChildren()
end

local grainTimer = 0
local conn = RunService.RenderStepped:Connect(function(dt)
	for _, c in ipairs(clouds) do
		c.x += c.speed * dt
		if c.x > W + 80 then c.x = -c.w end
		c.f.Position = UDim2.fromOffset(c.x, c.y)
	end
	for _, d in ipairs(drops) do
		d.y += d.v * dt
		d.x -= d.v * 0.2 * dt
		if d.y > H then d.y = -40 d.x = math.random(0, W + 200) end
		d.f.Position = UDim2.fromOffset(d.x, d.y)
	end
	for _, f in ipairs(fogs) do
		f.x += f.v * dt
		if f.x > 0 then f.x = -W * 0.6 end
		f.f.Position = UDim2.fromOffset(f.x, f.y)
	end
	for _, p in ipairs(dust) do
		p.y -= p.v * dt
		p.x += math.sin(os.clock() + p.sway) * 12 * dt
		if p.y < -10 then p.y = H + 10 end
		p.f.Position = UDim2.fromOffset(p.x, p.y)
		if dustOn and not p.shown then
			p.shown = true
			tw(p.f, 1.5, {BackgroundTransparency = math.random(40, 75) / 100})
		end
	end
	local t = os.clock()
	for _, r in ipairs(rays) do
		r.f.BackgroundTransparency = 0.935 + math.sin(t * 0.8 + r.ph) * 0.02
	end
	grainTimer += dt
	if grainTimer > 0.04 then
		grainTimer = 0
		for _, g in ipairs(grain) do
			g.Position = UDim2.fromOffset(math.random(0, W), math.random(0, H))
		end
	end
end)

task.spawn(function()
	local t0 = os.clock()
	local function waitUntil(t)
		while os.clock() - t0 < t do task.wait() end
	end

	tw(Root, 1.5, {GroupTransparency = 0})
	tw(BarTop, 1.6, {Size = UDim2.new(1, 0, 0.11, 0)}, Enum.EasingStyle.Quint)
	tw(BarBottom, 1.6, {Size = UDim2.new(1, 0, 0.11, 0)}, Enum.EasingStyle.Quint)
	tw(WorldScale, 9, {Scale = 1.09}, Enum.EasingStyle.Linear)

	waitUntil(2.6)
	task.spawn(strike, false)

	waitUntil(4.5)
	task.spawn(strike, true)

	waitUntil(5.4)
	tw(LogoGlow, 1.8, {BackgroundTransparency = 0.9, Size = UDim2.fromOffset(560, 300)})
	tw(LogoScale, 2.0, {Scale = 1}, Enum.EasingStyle.Quint)
	for i, L in ipairs(letters) do
		task.delay((i - 1) * 0.13, function()
			tw(L.l, 0.8, {TextTransparency = 0, Position = UDim2.fromOffset(L.x, 20)}, Enum.EasingStyle.Quint)
			tw(L.st, 0.8, {Transparency = 0.45})
		end)
	end

	waitUntil(6.9)
	Sweep.BackgroundTransparency = 0
	tw(Sweep, 1.0, {Position = UDim2.new(1.2, 0, 0.5, 0)}, Enum.EasingStyle.Quad, Enum.EasingDirection.InOut)

	waitUntil(7.4)
	dustOn = true
	tw(lineL, 1.2, {Size = UDim2.fromOffset(170, 2)}, Enum.EasingStyle.Quint)
	tw(lineR, 1.2, {Size = UDim2.fromOffset(170, 2)}, Enum.EasingStyle.Quint)
	tw(Sub, 1.2, {TextTransparency = 0.05})

	waitUntil(9.0)
	tw(Root, 1.0, {GroupTransparency = 1}, Enum.EasingStyle.Quad, Enum.EasingDirection.In)

	waitUntil(TOTAL)
	conn:Disconnect()
	pcall(function()

		IntroGui:Destroy()
	end)
	pcall(OnIntroFinished)
end)
